{{- /* A module named inside a walk arrives here already written out by its own shortcode (pm/mod, spec G8), and the Markdown renderer drops HTML it didn't make. So each name is held aside under a token while the walk's words are rendered, and put back after. */ -}} {{- $inner := .Inner -}} {{- $held := slice -}} {{- range $i, $h := findRE `(?s)|.*?.*?.*?` $inner -}} {{- $held = $held | append $h -}} {{- $inner = replace $inner $h (printf "MODNAMEHELD%dX" $i) 1 -}} {{- end -}} {{- $html := $inner | .Page.RenderString (dict "display" "block") -}} {{- range $i, $h := $held }}{{ $html = replace $html (printf "MODNAMEHELD%dX" $i) $h 1 }}{{ end }} {{ $html | safeHTML }}
